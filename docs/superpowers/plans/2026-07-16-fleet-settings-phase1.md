# Fleet Settings — Phase 1 Implementation Plan (read-only tree + store + test harness)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a behind-a-flag, read-only **Fleet** view of the agent/squad topology (expandable tree + legend + empty states), backed by a pure, unit-tested `FleetStore` data model — establishing the store, the round-trip-serialization foundation, and the project's first JS test harness.

**Architecture:** New `web/fleet/` client module. `store.js` is a pure, DOM-free, dual-export (browser global + CommonJS) module that turns the existing parsed agents config into a derived topology model and can serialize it back losslessly. `tree.js` renders that model as a read-only tree. A feature-flagged **Fleet** sub-tab in `settings.js` mounts the tree; the existing Agents/Squads sub-tabs are untouched. No Go/server changes.

**Tech Stack:** Vanilla ES5-style browser JS (no bundler, no ES modules in the browser — plain `<script>` globals, matching `app.js`/`settings.js`), Node 22 built-in `node --test` for unit tests, CSS `@import` partials, JSON i18n catalogues regenerated by `node scripts/build_i18n.mjs`.

## Global Constraints

- **No new runtime dependencies.** No npm packages, no bundler, no framework. Browser code is plain `<script>` globals (attach to `window.*`), exactly like `web/app.js` and `web/settings.js`. Node is a **dev/build tool only** (already used by `make i18n`).
- **No Go/server changes in Phase 1.** Reuse `GET /api/config/parsed/agent` (already returns full agent objects incl. `subagents`, and squads incl. `members`/`hidden`). The read model consumes `state.parsed["agent"].value`.
- **Round-trip fidelity is sacred.** `FleetStore.serialize(FleetStore.build(cfg))` must deep-equal `cfg` for every key present — `subagents`, `hidden`, squad `members`, `max_instances`, `resumable_sessions`, `max_tool_calls`, and any unknown keys. Unknown keys must pass through untouched.
- **Feature flag off by default.** The new view appears only when `localStorage["agent_toolkit_fleet_preview"] === "1"`. With the flag off, `settings.js` behavior is byte-identical to before.
- **i18n:** all user-facing strings are `tr("fleet.<key>")`; add keys to `web/i18n/en.json` + `fr.json` + `es.json` + `de.json`; regenerate with `make i18n`; bump the `?v=` on `settings.js` (and `i18n/locales.js` if changed) in `web/index.html`. Do not translate glossary nouns (Omnis, Squad, Skills, MCP, A2A, tool/config-key identifiers).
- **Legend glyphs (verbatim):** `◇` router, `⬢` squad, `⬡` leaderless squad, `★` leader, `•` member, `◆` sole agent, `↳` nested sub-agent, `⌂` shared badge prefix (rendered as `⌂N`).

```js
  function lc(s) { return String(s || "").toLowerCase(); }

  function build(cfg) {
    const agents = Array.isArray(cfg.agents) ? cfg.agents.slice() : [];
    const rawSquads = Array.isArray(cfg.squads) ? cfg.squads : [];
    const agentByName = new Map();
    agents.forEach(a => { if (a && a.name) agentByName.set(lc(a.name), a); });

    const routerSquadName = cfg.router_squad && lc(cfg.router_squad) !== "none"
      ? cfg.router_squad
      : (cfg.router_squad === undefined ? "omnis" : null);

    const squads = rawSquads.map(sq => {
      const leader = sq.leader === undefined ? "" : sq.leader;
      const leaderless = !leader || lc(leader) === "none";
      const isRouter = routerSquadName && lc(sq.name) === lc(routerSquadName);
      return {
        name: sq.name || "",
        leader: leaderless ? "" : leader,
        members: Array.isArray(sq.members) ? sq.members.slice() : [],
        hidden: !!sq.hidden,
        description: sq.description || "",
        kind: isRouter ? "router" : (leaderless ? "leaderless" : "squad"),
      };
    });

    return { agents, agentByName, squads, routerSquadName, raw: cfg };
  }
```

Add to `web/fleet/store.js` (inside the factory, before `return`), and add the three names to the returned object:

```js
  function sharedCounts(model) {
    const counts = new Map();
    const bump = n => { const k = lc(n); if (k) counts.set(k, (counts.get(k) || 0) + 1); };
    model.squads.forEach(sq => {
      if (sq.kind === "router") return;
      if (sq.leader) bump(sq.leader);
      sq.members.forEach(bump);
    });
    return counts;
  }

  function subagentNames(model, agent) {
    return Array.isArray(agent && agent.subagents) ? agent.subagents : [];
  }

  function collectUsed(model) {
    const used = new Set();
    model.squads.forEach(sq => {
      if (sq.leader) used.add(lc(sq.leader));
      sq.members.forEach(n => used.add(lc(n)));
    });
    // subagents (transitive) of any agent are "used"
    model.agents.forEach(a => subagentNames(model, a).forEach(n => used.add(lc(n))));
    return used;
  }

  function unusedAgents(model) {
    const used = collectUsed(model);
    return model.agents.filter(a =>
      a && a.name && a.enabled !== false && !used.has(lc(a.name)));
  }

  function pushSubtree(model, name, depth, seen, out) {
    const key = lc(name);
    if (seen.has(key)) return;           // cycle / repeat guard within one branch
    seen.add(key);
    const agent = model.agentByName.get(key);
    subagentNames(model, agent).forEach(sub => {
      out.push({ kind: "subagent", name: sub, agent: model.agentByName.get(lc(sub)), depth });
      pushSubtree(model, sub, depth + 1, seen, out);
    });
    seen.delete(key);
  }

  function treeNodes(model) {
    const out = [];
    const shared = sharedCounts(model);
    const orderedSquads = model.squads.slice().sort((a, b) => {
      if (a.kind === "router") return -1;
      if (b.kind === "router") return 1;
      return 0;
    });
    orderedSquads.forEach(sq => {
      out.push({ kind: sq.kind === "leaderless" ? "leaderless" : sq.kind === "router" ? "router" : "squad", name: sq.name, squad: sq, depth: 0 });
      if (sq.kind === "leaderless" || sq.kind === "router") {
        (sq.members[0] ? [sq.members[0]] : []).forEach(n => {
          out.push({ kind: "sole", name: n, agent: model.agentByName.get(lc(n)), depth: 1 });
          pushSubtree(model, n, 2, new Set(), out);
        });
        return;
      }
      if (sq.leader) {
        out.push({ kind: "leader", name: sq.leader, agent: model.agentByName.get(lc(sq.leader)), depth: 1, shared: shared.get(lc(sq.leader)) || 0 });
        pushSubtree(model, sq.leader, 2, new Set(), out);
      }
      sq.members.forEach(m => {
        out.push({ kind: "member", name: m, agent: model.agentByName.get(lc(m)), depth: 1, shared: shared.get(lc(m)) || 0 });
        pushSubtree(model, m, 2, new Set(), out);
      });
    });
    const unused = unusedAgents(model);
    if (unused.length) {
      out.push({ kind: "unused-header", name: "", depth: 0 });
      unused.forEach(a => out.push({ kind: "unused", name: a.name, agent: a, depth: 1 }));
    }
    return out;
  }
```

- [ ] **Step 3: Implement serialize + validators**

In `web/fleet/store.js`, replace the stub `serialize` and add the two validators; the key to round-trip is: **`build` keeps `raw` and does not mutate agents/squads**, so `serialize` reconstructs from `raw` structurally. Simplest correct implementation — serialize returns the original object graph shape:

```js
  function serialize(model) {
    // build() never mutates cfg and stores it as model.raw; the SquadView is a
    // derived read model. The canonical on-disk shape is exactly model.raw, so
    // round-trip is structural identity. (Phase 2 replaces this with a real
    // diff against edits; keeping identity here locks the contract.)
    return model.raw;
  }

  function subagentGraph(model) {
    const g = new Map();
    model.agents.forEach(a => g.set(lc(a.name), subagentNames(model, a).map(lc)));
    return g;
  }

  function wouldCreateCycle(model, fromName, toName) {
    const from = lc(fromName), to = lc(toName);
    if (from === to) return true;
    // Adding from→to creates a cycle iff `from` is reachable from `to`.
    const g = subagentGraph(model);
    const stack = [to], seen = new Set();
    while (stack.length) {
      const cur = stack.pop();
      if (cur === from) return true;
      if (seen.has(cur)) continue;
      seen.add(cur);
      (g.get(cur) || []).forEach(n => stack.push(n));
    }
    return false;
  }

  function validateSquad(sq) {
    const name = String(sq && sq.name || "").trim();
    if (!name) return { ok: false, error: "empty-name" };
    const leaderless = !sq.leader || lc(sq.leader) === "none";
    const members = Array.isArray(sq.members) ? sq.members : [];
    if (leaderless && members.length !== 1) return { ok: false, error: "leaderless-one-member" };
    if (!leaderless && members.length >= 2 && !sq.leader) return { ok: false, error: "needs-leader" };
    return { ok: true };
  }
```

```js
"use strict";
// FleetTree — read-only topology renderer for the Fleet settings view (Phase 1).
(function () {
  const esc = (s) => (typeof window !== "undefined" && window.escHtml)
    ? window.escHtml(s)
    : String(s == null ? "" : s).replace(/[&<>"]/g, c => ({ "&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;" }[c]));
  const t = (k, v) => (typeof window !== "undefined" && window.tr) ? window.tr(k, v) : k;

  const GLYPH = {
    router: "◇", squad: "⬢", leaderless: "⬡", leader: "★",
    member: "•", sole: "◆", subagent: "↳", unused: "·",
  };

  function legendHTML() {
    const items = [
      ["◇", t("fleet.legend.router")], ["⬢", t("fleet.legend.squad")],
      ["⬡", t("fleet.legend.leaderless")], ["★", t("fleet.legend.leader")],
      ["•", t("fleet.legend.member")], ["↳", t("fleet.legend.subagent")],
      ["⌂N", t("fleet.legend.shared")],
    ];
    return `<div class="fleet-legend">${items.map(([g, label]) =>
      `<span><code>${esc(g)}</code> ${esc(label)}</span>`).join("")}</div>`;
  }

  function rowHTML(n) {
    if (n.kind === "unused-header") {
      return `<div class="fleet-unused-header">${esc(t("fleet.unusedAgents"))}</div>`;
    }
    const glyph = GLYPH[n.kind] || "";
    const indent = 8 + n.depth * 16;
    const model = n.agent && n.agent.model_ref ? `<span class="fleet-sub">${esc(n.agent.model_ref)}</span>` : "";
    const badge = n.shared && n.shared > 1 ? `<span class="fleet-badge" data-tip="${esc(t("fleet.sharedTip", { n: n.shared }))}">⌂${n.shared}</span>` : "";
    let tag = "";
    if (n.kind === "router") tag = `<span class="fleet-tag">${esc(t("fleet.tag.router"))}</span>`;
    else if (n.kind === "leaderless") tag = `<span class="fleet-tag">${esc(t("fleet.tag.leaderless"))}</span>`;
    if (n.squad && n.squad.hidden) tag += `<span class="fleet-tag">${esc(t("fleet.tag.hidden"))}</span>`;
    return `<div class="fleet-row kind-${esc(n.kind)}" style="padding-left:${indent}px">
      <span class="fleet-glyph">${esc(glyph)}</span>
      <span class="fleet-name">${esc(n.name)}</span>${badge}${model}${tag}
    </div>`;
  }

  function render(container, model, _opts) {
    const nodes = window.FleetStore.treeNodes(model);
    const hasSquads = model.squads.some(s => s.kind !== "router");
    const body = hasSquads
      ? `<div class="fleet-tree">${nodes.map(rowHTML).join("")}</div>`
      : `<div class="fleet-empty">${esc(t("fleet.empty"))}</div>`;
    container.innerHTML = `<div class="fleet-view">${legendHTML()}${body}</div>`;
  }

  window.FleetTree = { render };
})();
```
